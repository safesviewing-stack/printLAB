/* ===================================================== ACCESO, LOGUEO Y SUBIDA STL DINÁMICA ===================================================== */ document.addEventListener("DOMContentLoaded", async function() { const accessButton = document.getElementById("analyzerAccessButton"); const fileInput = document.getElementById("stlFile"); const uploadContainer = document.getElementById("uploadContainer"); const selectedFile = document.getElementById("selectedFile"); const selectedFileName = document.getElementById("selectedFileName"); const selectedFileSize = document.getElementById("selectedFileSize"); if (!accessButton || !fileInput || !uploadContainer) { console.error("No se encontraron los elementos necesarios en el DOM."); return; } // 1. Inicialización de cliente Supabase let client = null; if (typeof window.supabaseClient !== "undefined" && window.supabaseClient) { client = window.supabaseClient; } else if (window.supabase && typeof window.supabase.auth !== "undefined") { client = window.supabase; } // 2. Comprobación del estado de sesión async function getCurrentSession() { try { if (client && client.auth && typeof client.auth.getSession === "function") { const result = await client.auth.getSession(); if (result && result.data && result.data.session) { return result.data.session; } } } catch (error) { console.warn("No se pudo verificar la sesión activa:", error); } return null; } // 3. Actualización dinámica de la interfaz basada en estados async function updateUI() { const session = await getCurrentSession(); if (!session) { // ESTADO: Sin sesión iniciada accessButton.textContent = "Inicia sesión para subir archivos"; accessButton.dataset.state = "logged-out"; selectedFile.classList.remove("show"); window.selectedSTLFile = null; } else { if (window.selectedSTLFile) { // ESTADO: Logueado y con archivo cargado (Listo para debitar y analizar) accessButton.textContent = "Iniciar análisis (1 crédito)"; accessButton.dataset.state = "has-file"; } else { // ESTADO: Logueado pero sin archivo cargado accessButton.textContent = "Seleccionar archivo STL"; accessButton.dataset.state = "logged-in-no-file"; selectedFile.classList.remove("show"); } } } await updateUI(); // Escuchar cambios de sesión de Supabase if (client && client.auth && typeof client.auth.onAuthStateChange === "function") { client.auth.onAuthStateChange(function() { updateUI(); }); } // 4. Redirección al login conservando la página de retorno function redirectToLogin() { const currentUrl = "analizador-stl.html"; const redirect = encodeURIComponent(currentUrl); window.location.href = "login.html?redirect=" + redirect; } // 5. Validación de restricciones del archivo STL function isValidSTL(file) { if (!file) return false; const name = file.name.toLowerCase(); return name.endsWith(".stl"); } function formatFileSize(bytes) { if (bytes < 1024) return bytes + " B"; if (bytes < 1024 * 1024) return (bytes / 1024).toFixed(1) + " KB"; return (bytes / (1024 * 1024)).toFixed(1) + " MB"; } // 6. Procesamiento del archivo tras selección o arrastre function handleFileSelection(file) { if (!file) return; if (!isValidSTL(file)) { alert("La plataforma requiere un archivo con formato geométrico .stl"); return; } const maxSize = 100 * 1024 * 1024; // Límite de 100 MB if (file.size > maxSize) { alert("El archivo excede el límite máximo permitido de 100 MB."); return; } window.selectedSTLFile = file; selectedFileName.textContent = "📄 " + file.name; selectedFileSize.innerHTML = ` ${formatFileSize(file.size)} Cambiar archivo `; selectedFile.classList.add("show"); // Lógica para permitir cambiar el archivo seleccionado const changeLink = document.getElementById("change-file-link"); if (changeLink) { changeLink.addEventListener("click", function(e) { e.stopPropagation(); window.selectedSTLFile = null; fileInput.value = ""; // Limpiar input físico updateUI(); }); } updateUI(); } // Escuchar entrada de archivos estándar fileInput.addEventListener("change", function() { if (fileInput.files && fileInput.files.length > 0) { handleFileSelection(fileInput.files[0]); } }); // 7. CONTROL DEL BOTÓN DE ACCIÓN (Lógica de estados) accessButton.addEventListener("click", async function(event) { event.preventDefault(); const session = await getCurrentSession(); // ESTADO 1: No ha iniciado sesión if (!session) { redirectToLogin(); return; } const state = accessButton.dataset.state; // ESTADO 2: Con sesión, pero sin archivo seleccionado (Abre el explorador de archivos) if (state === "logged-in-no-file") { fileInput.click(); return; } // ESTADO 3: Con sesión y archivo cargado (Ejecuta la deducción y análisis) if (state === "has-file") { accessButton.disabled = true; accessButton.textContent = "Verificando saldo..."; try { // Consultar el perfil de usuario para confirmar créditos const { data: profile, error: profileError } = await client .from("profiles") .select("credits") .eq("id", session.user.id) .single(); if (profileError) throw profileError; if (!profile || profile.credits < 1) { alert("Tu saldo de créditos es insuficiente para procesar el análisis. Redirigiendo a planes..."); window.location.href = "precios.html"; return; } accessButton.textContent = "Procesando geometría..."; // Ejecutar transacción segura del lado de la base de datos (RPC) const { data: success, error: rpcError } = await client.rpc("deduct_analysis_credit"); if (rpcError || !success) { throw new Error(rpcError?.message || "Ocurrió una inconsistencia en el cobro de créditos."); } // Simulación o transferencia de análisis exitosa alert("Análisis autorizado. Descontando 1 crédito de tu cuenta."); window.location.href = "informe-stl.html"; } catch (error) { console.error("Inconsistencia en el procesamiento:", error); alert("Error de validación: " + error.message); accessButton.disabled = false; updateUI(); } } }); // 8. Control de arrastrar y soltar (Drag and Drop) uploadContainer.addEventListener("dragover", function(event) { event.preventDefault(); uploadContainer.classList.add("drag-over"); }); uploadContainer.addEventListener("dragleave", function(event) { if (event.relatedTarget && uploadContainer.contains(event.relatedTarget)) { return; } uploadContainer.classList.remove("drag-over"); }); uploadContainer.addEventListener("drop", async function(event) { event.preventDefault(); uploadContainer.classList.remove("drag-over"); const session = await getCurrentSession(); if (!session) { redirectToLogin(); return; } const files = event.dataTransfer.files; if (files && files.length > 0) { handleFileSelection(files[0]); } }); });
